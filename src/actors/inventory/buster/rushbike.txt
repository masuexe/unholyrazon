const int WEPC_8RB_BIKEDIST = 24;

actor RushBikeWep : MegaManUBuster
{
    Weapon.AmmoUse 1
    Weapon.AmmoGive 100
    weapon.ammotype "RushBikeAmmo"
    Weapon.SlotNumber 1
    Inventory.Pickupmessage "$PU_RUSHBIKE"
    Obituary "$OB_RUSHBIKE"
    Tag "$TAG_RUSHBIKE"
	Weapon.AmmoUse2 1
	Weapon.AmmoGive2 3
	weapon.ammotype2 "BusterAmmo"
    inventory.icon "NULLICON"
    States
    {
    Spawn:
        RSHB A 1
        stop
    Ready:
        TNT1 A 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
        RSHB D 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_RUSHBIKE)
        TNT1 A 0 ACS_NamedExecuteWithResult("core_handleRushBike", false)
        TNT1 A 0 A_SpawnItemEx("RushBikeBikeFront", WEPC_8RB_BIKEDIST, 0, 0)
        TNT1 A 0 A_SpawnItemEx("RushBikeBikeBack", -WEPC_8RB_BIKEDIST, 0, 0)
        TNT1 A 0 A_GunFlash("BikeChooseState")
        TNT1 A 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
        Goto ReadyLoop
    TakeBusters:
        TNT1 A 0 A_GiveInventory("TakeBuster", 1)
        goto ReadyLoop
    ReadyLoop:
        TNT1 A 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
        loop
    Deselect:
        TNT1 A 0
        goto DeselectSwap
    Select:
        TNT1 A 0
        goto SelectSwap
    Altfire:
        TNT1 A 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
    Fire:
        TNT1 T 0 A_PlaySoundEx("assists/rushbikefire","Weapon")
        TNT1 T 0 A_FireCustomMissile("RushBikeMissile",0,0,0,0)
        TNT1 A 0 A_GiveInventory("RushBikeShooting_P", 1)
        TNT1 UW 1
        TNT1 T 0 A_JumpIfNoAmmo("NoAmmo")
        TNT1 T 0 A_PlaySoundEx("assists/rushbikefire","Weapon")
        TNT1 T 0 A_FireCustomMissile("RushBikeMissile",frandom(-3, 3),0,0,0)
        TNT1 A 0 A_GiveInventory("RushBikeShooting_P", 1)
        TNT1 VW 1
        TNT1 T 0 A_PlaySoundEx("assists/rushbikefire","Weapon")
        TNT1 T 0 A_FireCustomMissile("RushBikeMissile",frandom(-3, 3),0,0,0)
        TNT1 A 0 A_GiveInventory("RushBikeShooting_P", 1)
        TNT1 UW 1
        TNT1 T 0 A_PlaySoundEx("assists/rushbikefire","Weapon")
        TNT1 T 0 A_FireCustomMissile("RushBikeMissile",frandom(-3, 3),0,0,0)
        TNT1 A 0 A_GiveInventory("RushBikeShooting_P", 1)
        TNT1 VW 1
        TNT1 T 0 A_PlaySoundEx("assists/rushbikefire","Weapon")
        TNT1 T 0 A_FireCustomMissile("RushBikeMissile",frandom(-3, 3),0,0,0)
        TNT1 A 0 A_GiveInventory("RushBikeShooting_P", 1)
        TNT1 UW 1
        TNT1 TTTTT 2
        TNT1 A 0 A_Refire
        Goto ReadyLoop
    NoAmmo:
        TNT1 A 1
        TNT1 TTTTT 2
        Goto ReadyLoop

    BikeChooseState:
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 18, "RushBikeD")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 17, "RushBikeE")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 16, "RushBikeF")

        // The Curve
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 15, "RushBikeBlank")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 14, "RushBikeG")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 13, "RushBikeH")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 12, "RushBikeI")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 11, "RushBikeJ")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 10, "RushBikeK")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 9, "RushBikeL")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 8, "RushBikeM") // middle
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 7, "RushBikeN")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 6, "RushBikeO")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 5, "RushBikeP")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 4, "RushBikeQ")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 3, "RushBikeR")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 2, "RushBikeS")
        RSHB "#" 0 A_JumpIfInventory("RushBikeWepFrame", 1, "RushBikeBlank")
        goto RushBikeD

    RushBikeBlank:
        TNT1 A 1
        goto BikeChooseState
    RushBikeD:
        RSHB D 1
        goto BikeChooseState
    RushBikeE:
        RSHB E 1
        goto BikeChooseState
    RushBikeF:
        RSHB F 1
        goto BikeChooseState
    RushBikeG:
        RSHB G 1
        goto BikeChooseState
    RushBikeH:
        RSHB H 1
        goto BikeChooseState
    RushBikeI:
        RSHB I 1
        goto BikeChooseState
    RushBikeJ:
        RSHB J 1
        goto BikeChooseState
    RushBikeK:
        RSHB K 1
        goto BikeChooseState
    RushBikeL:
        RSHB L 1
        goto BikeChooseState
    RushBikeM:
        RSHB M 1
        goto BikeChooseState
    RushBikeN:
        RSHB N 1
        goto BikeChooseState
    RushBikeO:
        RSHB O 1
        goto BikeChooseState
    RushBikeP:
        RSHB P 1
        goto BikeChooseState
    RushBikeQ:
        RSHB Q 1
        goto BikeChooseState
    RushBikeR:
        RSHB R 1
        goto BikeChooseState
    RushBikeS:
        RSHB S 1
        goto BikeChooseState
    MegaBusterResume:
        goto ReadyLoop
    }
}

actor RushBikeMissile : BasicProjectile
{
    Radius 6
    Height 7
    damagetype "RushBike"
    Obituary "$OB_RUSHBIKE"
    damage (7 * (CheckClass("MegaManU", AAPTR_TARGET) ? UH_BUSTER_DAMAGE_SCALE : 1))
    speed 50
    States
    {
    Spawn:
        RSHB C 0
        RSHB C 0 A_SpawnItemEx("RushBikeMissilePuff", 0, 0, 0, 0, 0, 0, 0, SXF_ABSOLUTEPOSITION|SXF_WEPFXCOLOR)
        RSHB C 1
        wait
    }
}

actor RushBikeMissilePuff : BasicGraphicEffect
{
    +FORCEXYBILLBOARD
    var int user_x;
    var int user_y;
    var int user_z;
    States
    {
    SpawnFrame:
        HSNI O 0 A_ClientsideACSExecute("core_reloffsets_client", AAPTR_TARGET)
        HSNI OOPPQQRR 1 A_Warp(AAPTR_TARGET,user_x,user_y,user_z,angle,WARPF_COPYINTERPOLATION|WARPF_ABSOLUTEOFFSET|WARPF_ABSOLUTEANGLE)
        stop
    }
}

actor RushBikeBike : BasicShieldWarper
{
    Translation "192:192=4:4","198:198=42:42"
    speed 100
    States
    {
        Spawn:
            TNT1 A 0
            TNT1 A 0 ACS_NamedExecuteAlways("core_shieldwarper", 0, 1)
            TNT1 A 0 A_Jump(256, "Shield")
            Goto Shield
    }
}

actor RushBikeBikeFront : RushBikeBike
{
    States
    {
        Shield:
            RSHB T 0 ACS_NamedExecuteWithResult("core_copytranslation_client")
            RSHB T 0 A_Warp(AAPTR_TARGET, WEPC_8RB_BIKEDIST, 0, 0, 0, WARPF_NOCHECKPOSITION|WARPF_COPYINTERPOLATION)
            RSHB T 0 A_JumpIfInTargetInventory("RushBikeShooting_P", 1, "Shooting")
            RSHB T 1
            goto Shield
        Shooting:
            RSHB U 1
            goto Shield
    }
}
