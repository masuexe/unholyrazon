actor DuoFist_ScriptBar : ScriptBar {}

actor DuoFist : MegaManUBuster
{
    Weapon.AmmoUse 1
    Weapon.AmmoGive 3
    Weapon.SlotNumber 1
    Inventory.Pickupmessage "$PU_DUOFIST"
    Obituary "$OB_DUOFIST"
    Tag "$TAG_DUOFIST"
    weapon.ammotype "BusterAmmo"
    inventory.icon "NULLICON"
    Dropitem ""
    States
    {
        SpawnLoop:
            DUOF A 1
            stop
        Ready:
            DUOF C 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
            DUOF B 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
            Goto Ready2
        TakeBusters:
            TNT1 Q 0 A_GiveInventory("TakeBuster", 1)
            Goto Ready2
        Ready2:
            DUOF C 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_DUOFIST)
            DUOF C 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOFIST,TCLR_NONE)
            DUOF C 0 A_TakeInventory("WeaponCharge",999)
            DUOF C 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
            Goto Ready2+2
        Deselect:
            DUOF C 0
            goto DeselectSwap
        Select:
            DUOF C 0
            goto SelectSwap
        Fire:
            DUOF C 2
            DUOF C 0 A_Refire
            goto Fire1
        Altfire:
            DUOF C 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
        Fire1:
            DUOF M 0 A_JumpIfNoAmmo("Hold")
            DUOF M 0 A_PlaySoundEx("weapons/busters/duofistfire","Weapon")
            DUOF M 0 A_FireCustomMissile("DuoPunch",0,0,8,0)
            DUOF M 0 A_TakeInventory("WeaponCharge",999)
            DUOF M 1 A_TakeInventory("FistFlag",1)
            Goto FistWait
        FistWait:
            DUOF M 1
            DUOF M 0 A_GiveInventory("AdapterWaitCount",1)
            DUOF M 0 A_JumpIfInventory("AdapterWaitCount",20,"FistFinish")
            DUOF M 0 A_JumpIfInventory("FistFlag",1,"FistFinish")
            loop
        FistFinish:
            DUOF D 0 A_PlayWeaponSound("weapons/busters/duofistreturn")
            DUOF D 0 A_TakeInventory("FistFlag", 1)
            DUOF DC 3
            DUOF C 0 A_JumpIfInventory("AdapterWaitCount", 8, 2) //Adds a slight delay if fired at very close range
            DUOF C 4
            DUOF C 0 A_TakeInventory("AdapterWaitCount",999)
            DUOF C 0 A_Refire
            Goto Ready2+2
        Fire2:
            DUOF D 0 A_TakeInventory("WeaponCharge",999)
            DUOF D 2
            DUOF M 0 A_PlaySoundEx("weapons/busters/duofistfire2","Weapon")
            DUOF M 0 A_FireCustomMissile("MidFist",0,0,8,0)
            DUOF M 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOFIST,TCLR_NONE)
            DUOF M 16
            DUOF LDC 4
            DUOF B 0 A_Refire
            goto Ready2+2
        Fire3:
            DUOF E 0 A_TakeInventory("WeaponCharge",999)
            DUOF E 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOFIST,TCLR_NONE)
            DUOF EF 2
            DUOF G 0 A_PlaySoundEx("weapons/busters/duofistfire3","Weapon")
            DUOF G 0 A_FireCustomMissile("GigaFist",0,0,8,0)
            DUOF G 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOFIST,TCLR_NONE)
            DUOF G 16
            DUOF LDC 4
            DUOF B 0 A_Refire
            goto Ready2+2
        Hold:
            DUOF B 0 A_JumpIfInventory("WeaponCharge",18,"Charge")
            DUOF B 1 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
            DUOF B 0 A_Refire
            Goto Fire1
        Charge:
            DUOF B 0 A_JumpIfInventory("WeaponCharge",19,"Charge1")
            DUOF B 0 A_PlaySoundEx("weapons/busters/duofistcharge","Weapon")
            Goto Charge1
        Charge1:
            DUOF H 0 A_JumpIfInventory("WeaponCharge",35,"Charge2Start")
            DUOF H 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOCHARGE1,TCLR_CHARGE1)
            DUOF H 0 A_Refire("Charge1.a")
            Goto Fire2
        Charge1.a:
            DUOF B 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOFIST,TCLR_NONE)
            DUOF B 0 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
            DUOF B 0 A_Refire("Charge1")
            Goto Fire2
        Charge2Start:
            DUOF K 0 A_GunFlash("Sound.BusterCharge", GFF_NOEXTCHANGE)
            Goto Charge2
        Charge2:
            DUOF K 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOCHARGE2,TCLR_CHARGE2)
            DUOF K 0 A_Refire("Charge2.a")
            Goto Fire3
        Charge2.a:
            DUOF I 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOCHARGE3,TCLR_CHARGE3)
            DUOF I 0 A_Refire("Charge2.b")
            Goto Fire3
        Charge2.b:
            DUOF J 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_DUOFIST,TCLR_NONE)
            DUOF J 0 A_JumpIfInventory("WeaponCharge", 35, 3)
            DUOF J 0 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
            DUOF J 0 A_GunFlash
            DUOF J 0 A_JumpIfInventory("IsBot", 1, "BotFire")
            DUOF J 0 A_Refire("Charge2")
            Goto Fire3
        BotFire:
            DUOF J 0 A_Jump(16, "Fire3")
            DUOF J 0 A_Refire
            Goto Fire3
        MegaBusterResume:
            goto Ready2
    }
}

actor DuoPunch : BasicProjectile
{
    +SKYEXPLODE
    Damage (25 * (CheckClass("MegaManU", AAPTR_TARGET) ? UH_BUSTER_DAMAGE_SCALE : 1))
    Speed 32
    Radius 10
    Height 10
    damagetype "DuoFist1"
    Obituary "$OB_DUOFIST"
    States
    {
        Spawn:
            GIGF A 0
            GIGF A 0 ACS_NamedExecuteWithResult("core_drawhookshot", ACS_NamedExecuteWithResult("core_gettarget"), 32, 8, DHS_MAXDIST + 1024)
            GIGF A 0 A_GiveInventory("CutterFlag", 5)
        Spawn2:
            GIGF A 0
            GIGF A 2
            TNT1 A 0 A_SpawnItemEx("ThunderClawPegHelper")
            GIGF A 0 A_TakeInventory("CutterFlag", 1)
            GIGF A 0 A_JumpIfInventory("Cutterflag", 1, "Spawn2")
            GIGF A 0
            Goto Spawn3
        Spawn3:
            GIGF AA 1
            GIGF A 0 A_TakeInventory("CutterFlag", 999)
            Goto Death
        Death:
            TNT1 A 0 A_SpawnItemEx("ThunderClawPegHelper")
            TNT1 A 1 A_CustomMissile("DuoPunchReturn",0,0,0,0)
            stop
    }
}
