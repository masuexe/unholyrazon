actor TrebleBoost : MegaManUBuster
{
Weapon.AmmoUse 1
Weapon.AmmoGive 28
Weapon.SlotNumber 1
Inventory.Pickupmessage "$PU_TREBLEBOOST"
Obituary "$OB_TREBLEBOOST"
Tag "$TAG_TREBLEBOOST"
weapon.ammotype "BusterAmmo"
inventory.icon "NULLICON"
Dropitem ""
States
{
SpawnLoop:
TRBB I 1
stop
Ready:
    TREB B 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
TREB B 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
Goto Ready2
TakeBusters:
TNT1 A 0 A_GiveInventory("TakeBuster",1)
Goto Ready2
Ready2:
TREB B 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_TREBLEBOOST)
TREB B 0 A_SpawnItemEx("TrebleBoostWings",0,0,0,0,0,0,0,0)
Ready3:
TREB B 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
goto Ready3
Deselect:
TREB B 1 // one tic so that wings don't get confused
goto DeselectSwap
Select:
TREB B 0
goto SelectSwap
Fire:
TREB C 0 A_PlaySoundEx("weapons/busters/trebleboostfire","Weapon")
TREB C 0 A_FireCustomMissile("BoostShot",-15,0,8,0)
TREB C 0 A_FireCustomMissile("BoostShot",15,0,8,0)
TREB C 0 A_FireCustomMissile("BoostShot",0,0,8,0)
TREB DDDCCC 1
TREB B 6
goto Ready3
MegaBusterResume:
    goto Ready3
}
}

actor TrebleBoostWings : BasicShieldWarper
{
	States
	{
	Spawn:
		TNT1 A 0
		TNT1 A 0 ACS_NamedExecuteAlways("core_shieldwarper", 0, 1)
		TNT1 A 0 A_Jump(256, "Shield")
		Goto Shield
	Shield:
        TREB G 0 ACS_NamedExecuteWithResult("core_copytranslation_client",0)
        TREB G 1 A_Warp(AAPTR_TARGET, -12, 0, 18, 0, WARPF_INTERPOLATE|WARPF_COPYINTERPOLATION|WARPF_NOCHECKPOSITION)
        TREB G 0 A_JumpIfInTargetInventory("TrebleBoostJet", 1, "Smoke")
        loop
    Smoke:
        TREB G 0 A_SpawnItemEx("TrebleBoostTrail", 0, 18, 0, 0, 0, -5)
        TREB G 0 A_SpawnItemEx("TrebleBoostTrail", 0, -16, 0, 0, 0, -5)
        TREB G 0 ACS_NamedExecuteWithResult("core_copytranslation_client",0)
        TREB FE 1 bright A_Warp(AAPTR_TARGET, -12, 0, 18, 0, WARPF_INTERPOLATE|WARPF_COPYINTERPOLATION|WARPF_NOCHECKPOSITION)
        goto Shield
	}
}

actor TrebleBoostWatcher
{
Height 0
Radius 0
+NOINTERACTION
+NOGRAVITY
+MISSILE
States
{

TakeAmmo:

TNT1 A 0 ACS_NamedExecuteWithResult("core_flyplayer", ACS_NamedExecuteWithResult("core_gettarget", 0, true) + 1, false)

}
