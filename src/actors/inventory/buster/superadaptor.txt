actor SuperAdaptorWep : MegaManUBuster
{
	Weapon.AmmoUse 1
	Weapon.AmmoGive 3
	Weapon.SlotNumber 1
	Inventory.Pickupmessage "$PU_SUPERADAPTOR"
	Obituary "$OB_SUPERADAPTOR"
	Tag "$TAG_SUPERADAPTOR"
	weapon.ammotype "BusterAmmo"
	inventory.icon "NULLICON"
	Dropitem ""
	States
	{
	Spawn:
		WEA3 P 1
		stop
	Ready:
	    SAPT A 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
		SAPT A 0 A_GiveInventory("AdaptorFlag",1)
		TNT1 B 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
		goto Ready2
	TakeBusters:
		TNT1 Q 0 A_GiveInventory("TakeBuster", 1)
		goto Ready2
	Ready2:
		SAPT A 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_SUPERADAPTOR)
		SAPT A 0 A_SpawnItemEx("AdaptorWings",0,0,0,0,0,0,0,0)
	Ready3:
		SAPT A 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
		goto Ready3
	Deselect:
		SAPT A 1 A_StopSoundEx("Weapon")
		SAPT A 0 A_TakeInventory("AdaptorFlag",1)
		goto DeselectSwap
	Select:
		SAPT A 0
		goto SelectSwap
	Altfire:
	    SAPT A 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
	Fire:
		SAPT A 0 A_JumpIfNoAmmo("NoAmmo")
		SAPT A 0 A_PlaySoundEx("weapons/busters/superadaptorfire","Weapon")
		SAPT A 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTOR,TCLR_NONE)
		SAPT A 0 A_FireCustomMissile("MegaShot",0,0,8,0)
		SAPT A 0 A_TakeInventory("WeaponCharge",999)
		SAPT BC 3
		SAPT E 2
		SAPT B 0 A_Refire
		SAPT A 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTOR,TCLR_NONE)
		goto Ready3
	Hold:
		SAPT D 0 A_JumpIfInventory("WeaponCharge",41,"Charge2")
		SAPT D 0 A_JumpIfInventory("WeaponCharge",24,"Charge")
		SAPT A 1 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
		SAPT D 0 A_Refire
		goto Fire
	Charge:
		SAPT D 0 A_JumpIfInventory("WeaponCharge",41,"Charge2")
		SAPT D 0 A_JumpIfInventory("WeaponCharge",25,"Charge1")
		SAPT D 0 A_PlaySoundEx("weapons/busters/superadaptorcharge","Weapon")
		goto Charge1
	Charge1:
		SAPT A 0 A_JumpIfInventory("WeaponCharge",41,"Charge2Start")
		SAPT A 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTOR,TCLR_NONE)
		SAPT A 1 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
		SAPT A 0 A_Refire("Charge1.a")
		goto Fire
	Charge1.a:
		SAPT A 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTORCRG1,TCLR_PROTO_CHARGE1)
		SAPT D 1 //A_GiveInventory("AdapterCharge",1)
		SAPT A 0 A_Refire("Charge1")
		goto Fire
	Charge2Start:
		SAPT G 0 A_GunFlash("Sound.BusterCharge", GFF_NOEXTCHANGE)
	Charge2:
		SAPT G 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTORCRG2,TCLR_CHARGE2)
		SAPT A 0 A_Refire("Charge2.a")
		goto Fire2
	Charge2.a:
		SAPT H 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTORCRG3,TCLR_CHARGE3)
		SAPT A 0 A_Refire("Charge2.b")
		goto Fire2
	Charge2.b:
		SAPT K 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTOR,TCLR_NONE)
		SAPT A 0 A_Refire("Charge2.c")
		goto Fire2
	Charge2.c:
		SAPT K 0 A_JumpIfInventory("IsBot", 1, "BotFire")
		SAPT A 0 A_Refire("Charge2")
		goto Fire2
	BotFire:
		SAPT K 0 A_Jump(16, "Fire2")
		SAPT A 0 A_Refire
    Fire2:
		SAPT A 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_SUPERADAPTOR,TCLR_NONE)
        SAPT I 0 A_TakeInventory("WeaponCharge",999)
		SAPT I 0 A_GiveInventory("AdaptorArmWait",1)
        SAPT I 0 A_PlaySoundEx("weapons/busters/superadaptorfire2", "Weapon")
        SAPT I 0 A_FireCustomMissile("AdaptorFist",1,0,8,0)
		SAPT L 2
		SAPT I 8
        goto FistWait
    FistWait:
        TNT1 A 1 A_RailWait
        TNT1 A 0 A_JumpIfInventory("AdaptorArmWait", 1, "FistWait")
        TNT1 A 0 A_JumpIf(true, "FistFinish")
        loop
	FistFinish:
		SAPT I 0 A_PlaySoundEx("weapons/busters/superadaptorreturn", "Voice")
		goto Ready3
	NoFlash:
		TNT1 A 0
		stop
	NoAmmo:
		SAPT A 1
		goto Ready
	MegaBusterResume:
	    goto Ready3
	}
}
