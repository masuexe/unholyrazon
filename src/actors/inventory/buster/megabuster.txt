// MM8BDM v6c stock definition with MegaManU Buster damage scaling.
actor MegaBuster_MugshotColor : MugshotColor { args 192, 198 }

actor MegaBuster : BaseMM8BDMWep
{
Weapon.AmmoUse 1
Weapon.AmmoGive 3
Weapon.SlotNumber 1
Inventory.Pickupmessage "$PU_MEGABUSTER"
Obituary "$OB_MEGABUSTER"
Tag "$TAG_MEGABUSTER"
weapon.ammotype "BusterAmmo"
inventory.icon "NULLICON"
Dropitem ""
+WEAPON.WIMPY_WEAPON
+WEAPON.ALLOW_WITH_RESPAWN_INVUL
States
{
SpawnLoop:
WEAP X 1
loop
Ready:
BUST B 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_MEGABUSTER)
BUST B 1 A_WeaponReady
Goto Ready+1
Deselect:
BUST B 0
goto DeselectSwap
Select:
BUST B 0
goto SelectSwap
Fire:
BUST B 0 A_JumpIfNoAmmo("NoAmmo")
BUST B 0 A_PlaySoundEx("weapons/busters/megabusterfire","Weapon")
BUST B 0 A_FireCustomMissile("MegaShot",0,0,8,0)
BUST CD 3
BUST B 2
BUST B 0 A_Refire
goto Ready+1
NoAmmo:
BUST B 1
Goto Ready+1
Flash:
TNT1 A 1
stop
NoFlash:
TNT1 A 1
stop
}
}

actor BusterAmmo : Ammo
{
inventory.amount 1
inventory.maxamount 3
ammo.backpackamount 1
ammo.backpackmaxamount 3
}

actor MegaShot : BasicProjectile
{
+BRIGHT
damagetype "Buster"
Obituary "$OB_MEGABUSTER"
Speed 35
Damage (10 * (CheckClass("MegaManU", AAPTR_TARGET) ? UH_BUSTER_DAMAGE_SCALE : 1))
radius 10
height 5
States
{
Spawn:
TNT1 A 2
BUST A 1
wait
}
}
