actor JetAdaptorWep : MegaManUBuster
{
	Weapon.SlotNumber 1
	Obituary "$OB_JETADAPTOR"
	tag "$TAG_JETADAPTOR"
    Weapon.AmmoUse 0
    Weapon.AmmoGive 140
	weapon.ammotype "JetMeter"
	Weapon.AmmoUse2 1
	Weapon.AmmoGive2 3
	weapon.ammotype2 "BusterAmmo"
	inventory.icon "NULLICON"
	dropitem ""
	States
	{
		SpawnLoop:
			MMJT Z 1
			stop

		Deselect:
			MMJT D 1 // one tic so that wings don't get confused
			goto DeselectSwap
		Select:
			MMJT D 0
			goto SelectSwap

		Ready:
		    MMJT D 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
			MMJT D 0 A_JumpIfInventory("BusterGiven", 1, "TakeBusters")
			Goto Ready1
		TakeBusters:
			MMJT D 0 A_GiveInventory("TakeBuster", 1)
			Goto Ready1
		Ready1:
			MMJT D 0 ACS_NamedExecuteWithResult("core_weaponcolor", CLR_POWERFLIGHT)
			MMJT D 0 A_SpawnItemEx("JetAdaptorWings", -12, 0, 36)
			goto Ready3
		Ready3:
			MMJT D 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
			wait

		Fire:
		Fire1:
			MMJT E 0 A_PlaySoundEx("weapons/busters/jetadaptorfire", "Weapon")
			MMJT E 0 A_FireCustomMissile("JetAdaptorShot", 0, 0, 8, 0)
			MMJT EFG 1
			MMJT H 0 A_PlaySoundEx("weapons/busters/jetadaptorfire", "Weapon")
			MMJT H 0 A_FireCustomMissile("JetAdaptorShot", -frandom(1, 3), 0, 8, 0)
			MMJT HIJ 1
			MMJT K 0 A_PlaySoundEx("weapons/busters/jetadaptorfire", "Weapon")
			MMJT K 0 A_FireCustomMissile("JetAdaptorShot", frandom(1, 3), 0, 8, 0)
			MMJT KLM 1
			MMJT M 0 A_Refire("FireH1")
			goto FireH0
		Altfire:
		    MMJT D 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
		Fire2:
			MMJT E 0 A_PlaySoundEx("weapons/busters/jetadaptorfire", "Weapon")
			MMJT E 0 A_FireCustomMissile("JetAdaptorShot", 0, 0, 8, 0)
			MMJT EFG 1
			MMJT H 0 A_PlaySoundEx("weapons/busters/jetadaptorfire", "Weapon")
			MMJT H 0 A_FireCustomMissile("JetAdaptorShot", frandom(1, 3), 0, 8, 0)
			MMJT HIJ 1
			MMJT K 0 A_PlaySoundEx("weapons/busters/jetadaptorfire", "Weapon")
			MMJT K 0 A_FireCustomMissile("JetAdaptorShot", -frandom(1, 3), 0, 8, 0)
			MMJT KLM 1
			MMJT A 0 A_Refire("FireH2")
			goto FireH0

		FireH0:
			MMJT NOPPQQ 2
			goto Ready3
		FireH1:
			MMJT N 2 A_RailWait
			MMJT N 0 A_Refire(1)
			goto FireH0+1
			MMJT O 2
			MMJT O 0 A_Refire(1)
			goto FireH0+2
			MMJT P 2
			MMJT P 0 A_Refire(1)
			goto FireH0+3
			MMJT P 2
			MMJT P 0 A_Refire(1)
			goto FireH0+4
			MMJT Q 2
			MMJT Q 0 A_Refire(1)
			goto FireH0+5
			MMJT Q 2
			MMJT Q 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterRerollPending"), "RerollRefire")
			MMJT Q 0 A_Refire("Fire2")
			goto Ready3
		FireH2:
			MMJT N 2 A_RailWait
			MMJT N 0 A_Refire(1)
			goto FireH0+1
			MMJT O 2
			MMJT O 0 A_Refire(1)
			goto FireH0+2
			MMJT P 2
			MMJT P 0 A_Refire(1)
			goto FireH0+3
			MMJT P 2
			MMJT P 0 A_Refire(1)
			goto FireH0+4
			MMJT Q 2
			MMJT Q 0 A_Refire(1)
			goto FireH0+5
			MMJT Q 2
			MMJT Q 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterRerollPending"), "RerollRefire")
			MMJT Q 0 A_Refire("Fire1")
			goto Ready3
		RerollRefire:
			MMJT Q 0 A_Refire
			goto Ready3
		MegaBusterResume:
		    goto Ready3
	}
}

actor JetAdaptorWings : BasicShieldWarper
{
	Translation "192:192=4:4","198:198=42:42"
	States
	{
		Spawn:
			TNT1 A 0
			TNT1 A 0 ACS_NamedExecuteAlways("core_shieldwarper", 0, 1)
			TNT1 A 0 A_Jump(256, "Shield")
			Goto Shield
		Shield:
            MMJT A 0 A_StopSound(CHAN_VOICE)
			MMJT A 0 A_JumpIfInTargetInventory("JetBoostPower", 1, "Shield2Start")
			MMJT A 0 ACS_NamedExecuteWithResult("core_copytranslation_client")
			MMJT A 1 A_Warp(AAPTR_TARGET, -12, 0, 36, 0, WARPF_NOCHECKPOSITION|WARPF_COPYINTERPOLATION)
			loop
		Shield2Start:
			MMJT B 0 A_PlaySound("weapons/busters/jetadaptordash", CHAN_VOICE)
		Shield2:
			MMJT B 0 A_SpawnItemEx("BoosterFX", 0, -15, -12, 0, -2, -10, 0)
			MMJT B 0 A_SpawnItemEx("BoosterFX", 0, 15, -12, 0, 2, -10, 0)
			MMJT B 0 ACS_NamedExecuteWithResult("core_copytranslation_client")
			MMJT B 1 A_Warp(AAPTR_TARGET, -12, 0, 36, 0, WARPF_NOCHECKPOSITION|WARPF_COPYINTERPOLATION)
			MMJT B 0 A_JumpIfInTargetInventory("JetBoostPower", 1, 1)
			goto Shield
			MMJT B 0 ACS_NamedExecuteWithResult("core_copytranslation_client")
			MMJT B 1 A_Warp(AAPTR_TARGET, -12, 0, 36, 0, WARPF_NOCHECKPOSITION|WARPF_COPYINTERPOLATION)
			MMJT B 0 A_JumpIfInTargetInventory("JetBoostPower", 1, "Shield2")
			goto Shield
	}
}
