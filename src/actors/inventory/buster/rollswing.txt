actor RollSwing : MegaManUBuster
{
    Weapon.AmmoUse 1
    Weapon.AmmoGive 28
    Weapon.SlotNumber 1
    Inventory.Pickupmessage "$PU_ROLLSWING"
    Obituary "$OB_ROLLSWING"
    Tag "$TAG_ROLLSWING"
    weapon.ammotype "BusterAmmo"
    inventory.icon "NULLICON"
    Dropitem ""
    States
    {
    SpawnLoop:
        RSW1 A 1
        stop
    Ready:
        RSW1 H 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
        RSW1 H 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
        Goto Ready2
    TakeBusters:
        TNT1 A 0 A_GiveInventory("TakeBuster",1)
        Goto Ready2
    Ready2:
        RSW1 H 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_ROLLSWING)
    Ready3:
        RSW1 H 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
        loop
    Deselect:
        RSW1 H 0
        goto DeselectSwap
    Select:
        RSW1 H 0
        goto SelectSwap

    Fire:
        RSW1 H 0 A_JumpIfInventory("RollSwingSweep_P", 1, "Swing")
        RSW1 H 1
        goto Hold
    Hold:
    PreCharge:
        RSW1 H 0 A_JumpIfInventory("WeaponCharge",6,"ChargeStart")
        RSW1 H 1 A_GiveInventory("WeaponCharge",1)
        RSW1 H 0 A_JumpIfInventory("IsBot", 1, "BotPrechargeCheck")
        RSW1 H 0 A_Refire("PreCharge")
        Goto Swing
    ChargeStart:
        RSW1 H 0 ACS_NamedExecuteWithResult("core_wepcharge_rage",1,0) // gives one weapon charge only if rage rune
        RSW1 H 0 A_PlaySoundEx("weapons/busters/rollswingcharge", "Weapon")
    Charge:
        RSW1 V 0 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
        RSW1 V 0 A_JumpIfInventory("WeaponCharge", 45, "ChargeMaxStart")
        RSW1 V 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWINGCHARGE1,TCLR_CHARGE1)
        RSW1 V 0 A_Refire(1) goto ClearSwing
        RSW1 H 0 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
        RSW1 H 0 A_JumpIfInventory("WeaponCharge", 45, "ChargeMaxStart")
        RSW1 H 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWING,TCLR_NONE)
        RSW1 H 0 A_JumpIfInventory("IsBot", 1, "BotMidChargeCheck")
        RSW1 H 0 A_Refire("Charge") goto ClearSwing
    ChargeMaxStart:
        RSW1 H 0 A_GunFlash("Sound.BusterCharge", GFF_NOEXTCHANGE)
    ChargeMax:
        RSW1 W 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWINGCHARGE2,TCLR_CHARGE2)
        RSW1 W 0 A_Refire(1) goto SweepStart
        RSW1 X 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWINGCHARGE3,TCLR_CHARGE3)
        RSW1 X 0 A_Refire(1) goto SweepStart
        RSW1 H 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWING,TCLR_NONE)
        RSW1 H 0 A_JumpIfInventory("IsBot", 1, "BotMaxChargeCheck")
        RSW1 H 0 A_Refire("ChargeMax")
        goto SweepStart
	BotFire:
		RSW1 H 0 A_ClearRefire
        RSW1 H 0 A_Jump(32, "SweepStart")
		RSW1 H 0 A_Refire("ChargeMax")
        goto SweepStart

    Altfire:
        RSW1 H 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
        goto Swing
    AltHold:
        RSW1 H 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
        TNT1 A 0 A_ClearRefire
        goto Reraise

    ClearSwing:
        RSW1 I 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWING,TCLR_NONE)
        RSW1 I 0 A_TakeInventory("WeaponCharge")
    Swing:
        RSW1 I 0 A_TakeInventory("RollSwingSweep_P", 1)
        RSW1 I 0 A_GiveInventory("StopShield", 1)
        RSW1 IJ 2
        RSW1 J 0 A_TakeInventory("StopShield")
        RSW1 K 0 A_PlaySoundEx("weapons/busters/rollswingfire1", "Weapon")
        RSW1 K 0 A_FireCustomMissile("RollSwingLauncher1", 0, 0, 0, 0)
        RSW1 KLMN 1
        TNT1 A 5
        RSW1 O 0 A_PlaySoundEx("weapons/busters/rollswingfire2", "Weapon")
        RSW1 O 0 A_FireCustomMissile("RollSwingLauncher2", 0, 0, 0, 0)
        RSW1 OPQR 1
        TNT1 A 5
        goto Reraise

    SweepStart:
        RSW1 I 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_ROLLSWING,TCLR_NONE)
        RSW1 I 0 A_TakeInventory("WeaponCharge")
        RSW1 I 0 A_GiveInventory("StopShield", 1)
        RSW1 T 0 A_GiveInventory("RollSwingSweep_P", 1)
    SweepLoop:
        RSW1 T 0 A_JumpIfInventory("IsBot", 1, 2)
        RSW1 T 0 A_WeaponReady(WRF_NOSWITCH | (CheckClass("MegaManU") ? WRF_NOSECONDARY : 0))
        RSW1 T 1 A_GiveInventory("RollSwingSweep_CI", 1)
        RSW1 T 0 A_JumpIfInventory("IsBot", 1, 2)
        RSW1 T 0 A_WeaponReady(WRF_NOSWITCH | (CheckClass("MegaManU") ? WRF_NOSECONDARY : 0))
        RSW1 T 1 A_GiveInventory("RollSwingSweep_CI", 1)
        RSW1 U 0 A_JumpIfInventory("IsBot", 1, 2)
        RSW1 U 0 A_WeaponReady(WRF_NOSWITCH | (CheckClass("MegaManU") ? WRF_NOSECONDARY : 0))
        RSW1 U 1 A_GiveInventory("RollSwingSweep_CI", 1)
        RSW1 U 0 A_JumpIfInventory("IsBot", 1, 2)
        RSW1 U 0 A_WeaponReady(WRF_NOSWITCH | (CheckClass("MegaManU") ? WRF_NOSECONDARY : 0))
        RSW1 U 1 A_GiveInventory("RollSwingSweep_CI", 1)
        goto SweepCheck
    SweepCheck:
        TNT1 A 0 A_JumpIfInventory("IsBot", 1, "BotSweep")
        TNT1 A 0 A_JumpIfInventory("RollSwingSweep_P", 1, "SweepLoop")
        TNT1 A 0 A_GiveInventory("StopShield", 1)
        TNT1 A 3
        TNT1 A 0 A_TakeInventory("StopShield")
        goto Reraise
    SweepWait:
        RSW1 U 1
        goto SweepCheck

    Reraise:
        RSW1 JI 3
        goto Ready3

    BotPrechargeCheck:
        RSW1 H 0 A_JumpIf(CallACS("core_getPlayersInRadius", 120, PRF_CENTEROFMASS) > 1, "Swing")
        goto PreCharge
    BotMidChargeCheck:
        RSW1 H 0 A_JumpIf(random(1,100) < 33 && CallACS("core_getPlayersInRadius", 120, PRF_CENTEROFMASS) > 1, "ClearSwing")
        goto Charge
    BotMaxChargeCheck:
        RSW1 H 0 A_JumpIf(random(1,100) < 20 && CallACS("core_getPlayersInRadius", 120, PRF_CENTEROFMASS) > 1, "ClearSwing")
        RSW1 H 0 A_JumpIf(random(1,100) < 33, "SweepStart")
        goto ChargeMax
    BotSweep:
        RSW1 H 0 A_JumpIf(random(1,100) < 20 && CallACS("core_getPlayersInRadius", 120, PRF_CENTEROFMASS) > 1, "Fire")
        goto SweepCheck+1
    MegaBusterResume:
        goto Ready3
    }
}
