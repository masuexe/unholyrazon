actor ArrowBuster_ScriptBar : ScriptBar {}

actor ArrowBuster : LaserBuster
{
Inventory.Pickupmessage "$PU_ARROWBUSTER"
Obituary "$OB_ARROWBUSTER"
Tag "$TAG_ARROWBUSTER"
States
{
SpawnLoop:
WEA4 L 1
stop
Ready:
    CHBU K 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
CHBU K 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
Goto Ready2
TakeBusters:
TNT1 Q 0 A_GiveInventory("TakeBuster", 1)
Goto Ready2
Ready2:
CHBU K 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_MEGABUSTER)
CHBU K 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGABUSTER,TCLR_NONE)
CHBU K 0 A_TakeInventory("WeaponCharge",999)
CHBU K 2 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
Goto Ready2+2
Altfire:
    CHBU K 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
Fire:
CHBU K 0 A_JumpIfNoAmmo("Hold")
CHBU K 0 A_PlaySoundEx("weapons/busters/arrowbusterfire","Weapon")
CHBU K 0 A_FireCustomMissile("MegaShot",0,0,8,0)
CHBU K 1 A_TakeInventory("WeaponCharge",999)
CHBU LM 3
CHBU K 1
CHBU K 0 A_Refire
goto Ready2+2
Fire2:
CHBU K 0 A_TakeInventory("WeaponCharge",999)
CHBU L 0 A_PlaySoundEx("weapons/busters/arrowbusterfire2","Weapon")
CHBU K 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGABUSTER,TCLR_NONE)
CHBU L 0 A_FireCustomMissile("MidShot",0,0,8,0)
CHBU LM 3
CHBU K 0 A_Refire
goto Ready2+2
Fire3:
CHBU K 0 A_TakeInventory("WeaponCharge",999)
CHBU K 0 A_PlaySoundEx("weapons/busters/arrowbusterfire3","Weapon")
CHBU K 0
CHBU L 0 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGABUSTER,TCLR_NONE)
CHBU L 0 A_FireCustomMissile("ArrowShot",0,0,8,0)
CHBU LM 3
CHBU L 9
CHBU K 0 A_Refire
goto Ready2+2
BotFire:
CHBU K 0 A_Jump(16, "Fire3")
CHBU H 0 A_Refire("Charge2")
Goto Fire3
Hold:
//LBST B 0 A_JumpIfInventory("WeaponCharge",34,"Charge2Start")
CHBU K 0 A_JumpIfInventory("WeaponCharge",18,"Charge")
CHBU K 1 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
CHBU K 0 A_Refire
Goto Ready2+2
Charge:
//LBST B 0 A_JumpIfInventory("WeaponCharge",34,"Charge2Start")
CHBU K 0 A_JumpIfInventory("WeaponCharge",19,"Charge1")
CHBU K 0 A_PlaySoundEx("weapons/busters/laserbustercharge","Weapon")
Goto Charge1
Charge1:
CHBU K 0 A_JumpIfInventory("WeaponCharge",35,"Charge2Start")
CHBU N 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGACHARGE1,TCLR_CHARGE1)
CHBU H 0 A_Refire("Charge1.a")
Goto Fire2
Charge1.a:
CHBU K 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGABUSTER,TCLR_NONE)
CHBU K 0 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
CHBU H 0 A_Refire("Charge1")
Goto Fire2
Charge2Start:
CHBU K 0 A_GunFlash("Sound.BusterCharge", GFF_NOEXTCHANGE)
Goto Charge2
Charge2:
CHBU K 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGACHARGE2,TCLR_CHARGE2)
CHBU H 0 A_Refire("Charge2.a")
Goto Fire3
Charge2.a:
CHBU O 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGACHARGE3,TCLR_CHARGE3)
CHBU H 0 A_Refire("Charge2.b")
Goto Fire3
Charge2.b:
CHBU P 1 ACS_NamedExecuteWithResult("core_chargecolor",CLR_MEGABUSTER,TCLR_NONE)
CHBU K 0 A_JumpIfInventory("IsBot", 1, "BotFire")
CHBU K 0 A_JumpIfInventory("WeaponCharge", 35, 2)
CHBU K 0 A_GiveInventory("WeaponCharge", 1 + CallACS("core_checkrune", RUNE_RAGE))
CHBU K 0 A_Refire("Charge2")
Goto Fire3
MegaBusterResume:
    goto Ready2
}
}
