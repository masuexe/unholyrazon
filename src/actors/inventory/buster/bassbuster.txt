actor BassBuster_MugshotColor : MugshotColor { args 217, 95 }

actor BassBuster : MegaManUBuster
{
Weapon.AmmoUse 1
Weapon.AmmoGive 3
Weapon.SlotNumber 1
Inventory.Pickupmessage "$PU_BASSBUSTER"
Obituary "$OB_BASSBUSTER"
Tag "$TAG_BASSBUSTER"
weapon.ammotype "BusterAmmo"
inventory.icon "NULLICON"
Dropitem ""
States
{
SpawnLoop:
WEA3 A 1
stop
Ready:
    BASB B 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
BASB B 0 A_JumpIfInventory("BusterGiven",1,"TakeBusters")
Goto Ready2
TakeBusters:
TNT1 Q 0 A_GiveInventory("TakeBuster", 1)
Goto Ready2
Ready2:
BASB B 0 ACS_NamedExecuteWithResult("core_weaponcolor",CLR_BASSBUSTER)
BASB B 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
Goto Ready2+1
Deselect:
BASB B 0
goto DeselectSwap
Select:
BASB B 0
goto SelectSwap
Altfire:
BASB B 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
Fire:
BASB D 0 A_PlaySoundEx("weapons/busters/bassbusterfire","Weapon")
BASB D 0 A_GiveInventory("BassBuster_Firing_P", 1)
BASB D 0 A_GiveInventory("BassBuster_Fire_P")
Anim:
BASB C 2
BASB DB 1
BASB B 0 A_Refire
Cooldown:
BASB B 2 A_TakeInventory("BassBuster_Firing_P")
BASB B 2 A_SpawnItemEx("BassBuster_RageWatcher_H")
goto Ready2+1
MegaBusterResume:
    goto Ready2
}
}

actor BassBuster_RageWatcher_H : BasicWatcher
{
+NOTIMEFREEZE
+SERVERSIDEONLY
States
{
Spawn:
TNT1 A 0
TNT1 A 1 A_TakeFromTarget("BassBusterRage", 1 + CallACS("core_checkrune", RUNE_RAGE, AAPTR_TARGET))
TNT1 A 0 A_JumpIf(!CallACS("core_targetexists"), "Death")
TNT1 A 0 A_JumpIfInTargetInventory("BassBuster_Firing_P", 1, "Cease")
TNT1 A 0 A_JumpIfInTargetInventory("BassBusterRage", 1, 1)
stop
TNT1 A 1 A_JumpIfInTargetInventory("BassBuster_Firing_P", 1, "Cease")
loop
Death:
TNT1 A 0 A_TakeFromTarget("BassBusterRage")
Cease:
TNT1 A 0
stop
}
}
