actor PowerAdaptorWep : MegaManUBuster
{
    Weapon.SlotNumber 1
    Obituary "$OB_POWERADAPTOR"
    tag "$TAG_POWERADAPTOR"
    Weapon.AmmoUse 0
    Weapon.AmmoGive 0
	weapon.ammotype "PowerMeter"
	Weapon.AmmoUse2 1
	Weapon.AmmoGive2 3
	weapon.ammotype2 "BusterAmmo"
    inventory.icon "NULLICON"
    dropitem ""
    States
    {
        SpawnLoop:
            POWW Z 1
            stop

        Deselect:
            POWF A 0 A_TakeInventory("PowerMeter")
            goto DeselectSwap
        Select:
            POWF A 0
            goto SelectSwap

        Ready:
            POWF A 0 ACS_NamedExecuteWithResult("uh_MegaCommitBuster")
            POWF A 0 A_JumpIfInventory("BusterGiven", 1, "TakeBusters")
            Goto Ready2
        TakeBusters:
            POWF A 0 A_GiveInventory("TakeBuster", 1)
            Goto Ready2
        Ready2:
            POWF A 0 ACS_NamedExecuteWithResult("core_weaponcolor", CLR_POWERFLIGHT)
            POWF A 0 A_SpawnItemEx("PowerShoulderF", 0.1, 0, 30)
            POWF A 0 A_SpawnItemEx("PowerShoulderB", -0.1, 0, 30)
        Ready3:
            POWF A 0 A_TakeInventory("PowerAdaptorScore_P")
            POWF A 0 A_TakeInventory("PowerAdaptorSwing_P")
            POWF A 0 A_TakeInventory("PowerAdaptorStrike_P")
            POWF A 0 A_TakeInventory("PowerAdaptorStrike_P")
            POWF A 1 A_WeaponReady(CheckClass("MegaManU") ? CallACS("uh_MegaBusterReadyFlags", WRF_NOSECONDARY, WRF_ALLOWRELOAD) : 0)
            wait

        Fire:
        WindupA:
            POWF A 1
            POWF A 0 A_Refire(1) Goto Fire1
            POWF A 0 A_GiveInventory("PowerMeter", 1 + CallACS("core_checkrune", RUNE_RAGE))
            POWF A 0 A_JumpIfInventory("PowerMeter", 5, 1)
            loop

            POWF B 0 A_PlaySoundEx("weapons/busters/poweradaptorcharge1","Weapon")
        WindupB:
            POWF B 1
            POWF B 0 A_Refire(1) Goto Fire1
            POWF B 0 A_GiveInventory("PowerMeter", 1 + CallACS("core_checkrune", RUNE_RAGE))
            POWF B 0 A_JumpIfInventory("PowerMeter", 1*10+5, 1)
            loop

            POWF C 0 A_PlaySoundEx("weapons/busters/poweradaptorcharge2","Weapon")
        WindupC:
            POWF [ 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT1, TCLR_POWERFLIGHT1)
            POWF C 0 A_Refire(1) Goto Fire1
            POWF C 0 A_GiveInventory("PowerMeter", 1 + CallACS("core_checkrune", RUNE_RAGE))
            POWF C 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT, TCLR_DEFAULT)
            POWF C 0 A_Refire(1) Goto Fire1
            POWF C 0 A_GiveInventory("PowerMeter", 1 + CallACS("core_checkrune", RUNE_RAGE))
            POWF C 0 A_JumpIfInventory("PowerMeter", 2*10+5, 1)
            loop

            POWF D 0 A_PlaySoundEx("weapons/busters/poweradaptorcharge3","Weapon")
        WindupD:
            POWF ] 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT2, TCLR_POWERFLIGHT2)
            POWF D 0 A_Refire(1) Goto Fire1
            POWF D 0 A_GiveInventory("PowerMeter", 1 + CallACS("core_checkrune", RUNE_RAGE))
            POWF \ 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT1, TCLR_POWERFLIGHT1)
            POWF D 0 A_Refire(1) Goto Fire1
            POWF D 0 A_GiveInventory("PowerMeter", 1 + CallACS("core_checkrune", RUNE_RAGE))
            POWF D 0 A_JumpIfInventory("PowerMeter", 3*10+5, 1)
            loop

        Charge2_Snd:
            POWF D 0 A_PlaySoundEx("weapons/busters/poweradaptorcharge4","Weapon")
            TNT1 A 0 A_GunFlash("Flash.Charge", 1)
        Charge2:
            TNT1 A 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT3, TCLR_POWERFLIGHT3)
            TNT1 A 0 A_Refire(1)
            Goto Fire2
            TNT1 A 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT4, TCLR_POWERFLIGHT4)
            TNT1 A 0 A_Refire(1)
            Goto Fire2
            TNT1 A 1 ACS_NamedExecuteWithResult("core_chargecolor", CLR_POWERFLIGHT, TCLR_DEFAULT)
            TNT1 A 0 A_JumpIfInventory("IsBot", 1, "BotFire")
            TNT1 A 0 A_Refire("Charge2")
            Goto Fire2
        BotFire:
            TNT1 A 0 A_Jump(16, "Fire2")
            TNT1 A 0 A_Refire("Charge2")
            Goto Fire2

        Altfire:
        Althold:
            POWF A 0 A_JumpIf(CheckClass("MegaManU") && CallACS("uh_MegaBusterTryReroll"), "MegaReroll")
            POWF E 0
            goto Fire1

        Fire1:
            POWF E 0 A_JumpIfInventory("PowerAdaptorSwing_P", 1, "RightJabSwing")
        LeftJabSwing:
            POWF E 0 A_GiveInventory("PowerAdaptor_FireReset_CI")
            POWF E 0 A_PlaySoundEx("weapons/busters/poweradaptorjab","Weapon")
            POWF E 0 A_FireCustomMissile("PowerAdaptorJab",0,0,-8,0)
            POWF EF 1
            POWF GGGGG 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "LeftJabStrike")
            POWF HHHIIIJJJ 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "LeftJabStrike")
            POWF J 0 A_GiveInventory("PowerAdaptorSwing_P", 1)
            goto ReturnNeutral
        LeftJabStrike:
            POWF GQGGGG 1
            POWF HHHIIIJJJ 1
            POWF J 0 A_GiveInventory("PowerAdaptorSwing_P", 1)
            goto ReturnNeutral

        RightJabSwing:
            POWF K 0 A_TakeInventory("PowerAdaptorSwing_P", 1)
            POWF K 0 A_GiveInventory("PowerAdaptor_FireReset_CI")
            POWF K 0 A_PlaySoundEx("weapons/busters/poweradaptorjab","Weapon")
            POWF K 0 A_FireCustomMissile("PowerAdaptorJab",0,0,8,0)
            POWF KL 1
            POWF MMMMM 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightJabStrike")
            POWF NNNOOOPPP 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightJabStrike")
            goto ReturnNeutral
        RightJabStrike:
            POWF MRMMMM 1
            POWF NNNOOOPPP 1
            goto ReturnNeutral
        Fire2:
        RightStraight:
            POWF S 0 A_GiveInventory("PowerAdaptor_FireReset_CI")
            POWF S 0 A_PlaySoundEx("weapons/busters/poweradaptorstraight","Weapon")
            POWF S 0 A_FireCustomMissile("PowerAdaptorStraight",0,0,8,0)
            PWOF S 0 A_SpawnItemEx("PowerAdaptorReflect",0,8,0+8,0,0,0)
            PWOF S 0 A_SpawnItemEx("PowerAdaptorReflect2",cos(pitch)*40,8,-sin(pitch)*40+8,0,0,0)
            PWOF S 0 A_SpawnItemEx("PowerAdaptorReflect3",cos(pitch)*80,8,-sin(pitch)*80+8,0,0,0)

            POWF S 0 A_SetAngle(angle + 1)
            POWF S 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF T 0 A_SetAngle(angle + 1)
            POWF T 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF U 0 A_SetAngle(angle + 1)
            POWF U 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")

            POWF V 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF V 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF X 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF X 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF Y 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF Y 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF Z 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            POWF Z 1 A_JumpIfInventory("PowerAdaptorStrike_P", 1, "RightStraightStrike")
            goto ReturnNeutral
        RightStraightStrike:
            POWF V 0 A_GiveInventory("PowerAdaptorTremor_CI", 1)
            POWF VWVWVVV 1 A_JumpIfInventory("PowerAdaptorScore_P", 1, "RightStraightScore")
            POWF XYZ 3
            goto ReturnNeutral
        RightStraightScore:
            POWF W 0 A_GiveInventory("PowerAdaptorReflectSound_CI", 1)
            POWF WVWVWVVVV 1
            POWF V 0 A_TakeInventory("PowerAdaptorScore_P")
            POWF XYZ 3
            goto ReturnNeutral

        ReturnNeutral:
            TNT1 A 3 A_Refire
            POWF DCBA 2 A_Refire
            goto Ready3


        Flash.Charge:
            TNT1 A 0 A_PlaySoundEx("weapons/busters/poweradaptorcharge", "SoundSlot5")
            TNT1 A 9
            TNT1 A 0 A_JumpIfInventory("PowerMeter", 1, "Flash.Charge")
            stop
        MegaBusterResume:
            goto Ready3
    }
}

// Projectiles and FX
actor PowerAdaptorJab : BasicFastProjectile
{
    seesound "weapons/busters/poweradaptorshock"
    +THRUSPECIES
    +BRIGHT
    +FORCEXYBILLBOARD

    Damage (12 * (CheckClass("MegaManU", AAPTR_TARGET) ? UH_BUSTER_DAMAGE_SCALE : 1))
    damagetype "PowerJab"
    Obituary "$OB_POWERADAPTOR"

    radius 10
    height 10
    Speed 60

    ReactionTime 9

    States
    {
    Spawn:
        POWW A 0
        POWW CC 1 A_CountDown
        MMFX CDE 1 A_CountDown
        MMFX E 0 A_ChangeFlag("FORCEXYBILLBOARD", false)
        POWW A 0 A_Jump(256, 1, 2, 3, 4)
    SpawnL:
        POWW AABB 1 A_CountDown
        loop
    Death:
        MMFX E 0 A_Explode(15 * (CheckClass("MegaManU", AAPTR_TARGET) ? UH_BUSTER_DAMAGE_SCALE : 1), 60, 0, 0, 20)
    XDeath:
        MMFX E 0 A_JumpIf(Reactiontime >= 8, "Melee")
        MMFX E 0 A_ChangeFlag("FORCEXYBILLBOARD", true)
        MMFX EDCB 2
        POWW C 2
        stop
    Melee:
        MMFX E 0 A_GiveToTarget("PowerAdaptorStrike_P", 1)
        goto XDeath+1
    }
}

// Reflection stuff
actor PowerAdaptorReflect : BasicShieldHitbox
{
    reactiontime 5
    height 40
    radius 20

    +REFLECTIVE
    var int user_smartReflective;
    var int user_scriptedReflect;

    Meleerange 0 // Warp X
    Accuracy 8 // Warp Y
    Mass 8 // Warp Z
    States
    {
    Spawn:
        EBAL W 0
        EBAL W 0 A_JumpIfTargetInLOS("SpawnFrame", 0)
        goto Death
    SpawnFrame:
        EBAL W 0 ACS_NamedExecuteWithResult("core_followpitchyaw")
    SpawnLoop:
        EBAL W 0 A_JumpIf(ACS_NamedExecuteWithResult("core_targetexists")==0,"Death2")
        EBAL W 0 A_JumpIfInTargetInventory("StopShield",1,"SpawnStop")
        EBAL W 0 A_Warp(2,Meleerange*cos(pitch),Accuracy,Meleerange*-sin(pitch)+Mass,0,24)
        EBAL W 1 A_Countdown
        loop
    Pain:
        TNT1 A 0 A_GiveToTarget("PowerAdaptorReflectSound_CI", 1)
        TNT1 A 0 A_SpawnItemEx("PowerAdaptorReflectFX", 0, -accuracy, 28)
        goto Spawn
    }
}
