actor SpareBodyItem : BaseMM8BDMUseItem 10178
{
	//$Category MM8BDM-Assists/Utility
	//$Title Spare Body
	//$Sprite PSLDV0
	inventory.amount 1
	inventory.maxamount 1
	inventory.pickupmessage "$PU_SPAREBODY"
	Tag "$TAG_SPAREBODY"
	inventory.icon "SPARBODI"
	Inventory.PickupSound "item/1up"
	scale 2.0
	+COUNTITEM
	+INVBAR
	States
	{
		SpawnLoop:
			PSLD V -1
			loop
		Use:
			PSLD V 0 A_JumpIf(CallACS("uh_SpareBodyActivate"), "UseSuccess")
			fail
		UseSuccess:
			PSLD V 0 A_PlaySoundEx("item/refill", "Voice")
			stop
	}
}
