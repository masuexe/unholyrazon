// MM8BDM v6c stock definition with MegaManU input and upgrade support.
actor MegaManUBusterUpgrade : BaseMM8BDMBusterUpgrade
{
    Inventory.ForbiddenTo ""
    States
    {
    Pickup:
        TNT1 A 0 A_JumpIf(CheckClass("CustWepClassBase", AAPTR_DEFAULT, true) && !CheckClass("MegaManU"), "PickupRejected")
        goto Super::Pickup
    PickupRejected:
        TNT1 A 0
        fail
    }
}

actor BassUpgrade : MegaManUBusterUpgrade 10155
{
//$Category MM8BDM-Weapons/Buster
//$Title Bass Buster Upgrade
//Inventory.RespawnTics 350
//$Sprite WEA3A0
inventory.amount 1
inventory.maxamount 1
inventory.pickupmessage "$PU_BASSBUSTER"
Tag "$TAG_BASSUPGRADE"
inventory.icon "BASSUP"
Inventory.PickupSound "item/1up"
Inventory.RespawnTics 350
scale 2.0
+COUNTITEM
+INVBAR
states
{
SpawnLoop:
WEA3 A 1
loop
Use:
    TNT1 A 0 A_JumpIf(!CheckClass("MegaManU"), "NativeUse")
    TNT1 A 0 A_JumpIf(CallACS("uh_MegaInstallBuster", 6), "Installed")
InstallFailed:
    TNT1 A 0
    fail
Installed:
    TNT1 A 0 A_PlaySoundEx("item/refill", "Voice")
    stop
NativeUse:
    TNT1 A 0 A_JumpIf(CheckClass("CustWepClassBase", AAPTR_DEFAULT, true), "InstallFailed")
WEA2 Q 0 A_PlaySoundEx("item/refill","Voice")
TRBB I 0 A_GiveInventory("BusterGiven",1)
WEA3 A 0 A_GiveInventory("BassBuster",1)
WEA3 A 0 A_SelectWeapon("BassBuster")
stop
}
}
